<script>
(()=>{
  const canvas=document.getElementById('gameCanvas'),ctx=canvas.getContext('2d');
  const MAPS=[
    {id:'mixed',name:'混合',icon:'🌍'},
    {id:'road',name:'馬路',icon:'🚗'},
    {id:'river',name:'河流',icon:'🌊'},
    {id:'rail',name:'鐵軌',icon:'🚄'}
  ];
  const CHARACTERS=[
    {id:'chicken',name:'小雞',price:0},
    {id:'penguin',name:'企鵝',price:30},
    {id:'rabbit',name:'兔子',price:60},
    {id:'cat',name:'貓咪',price:100}
  ];
  const STORAGE_KEY='crossy_friends_save_v3';
  let saveData={bestScore:0,coins:0,unlocked:['chicken'],selected:'chicken',mapMode:'mixed'};
  try{
    const raw=localStorage.getItem(STORAGE_KEY);
    if(raw){
      const p=JSON.parse(raw);
      saveData={...saveData,...p};
      if(!saveData.unlocked.includes('chicken'))saveData.unlocked.push('chicken');
      if(!CHARACTERS.some(c=>c.id===saveData.selected))saveData.selected='chicken';
    }
  }catch(e){}
  function persistSave(){try{localStorage.setItem(STORAGE_KEY,JSON.stringify(saveData))}catch(e){}}

  let soundEnabled=true,audioCtx=null;
  function ensureAudio(){
    if(!audioCtx){const AC=window.AudioContext||window.webkitAudioContext;if(AC)audioCtx=new AC()}
    if(audioCtx&&audioCtx.state==='suspended')audioCtx.resume();
  }
  function playTone(f1,f2,dur,type='sine',vol=.15){
    if(!soundEnabled)return;ensureAudio();if(!audioCtx)return;
    const o=audioCtx.createOscillator(),g=audioCtx.createGain();
    o.type=type;o.frequency.setValueAtTime(f1,audioCtx.currentTime);
    if(f2)o.frequency.exponentialRampToValueAtTime(f2,audioCtx.currentTime+dur);
    g.gain.setValueAtTime(vol,audioCtx.currentTime);g.gain.exponentialRampToValueAtTime(.001,audioCtx.currentTime+dur);
    o.connect(g);g.connect(audioCtx.destination);o.start();o.stop(audioCtx.currentTime+dur);
  }
  const sfx={
    jump:()=>playTone(260,520,.11,'triangle',.16),
    coin:()=>{playTone(880,1320,.09,'sine',.16);setTimeout(()=>playTone(1320,1760,.14,'sine',.14),60)},
    splash:()=>playTone(220,65,.35,'sawtooth',.18),
    crash:()=>playTone(140,42,.32,'square',.22),
    train:()=>playTone(620,580,.18,'sawtooth',.08),
    buy:()=>{playTone(523,659,.1,'triangle',.18);setTimeout(()=>playTone(784,1046,.18,'triangle',.18),90)}
  };

  let W=0,H=0,DPR=1;const TILE=52,COLS=13,HALF_COLS=6;
  function resize(){
    DPR=Math.min(window.devicePixelRatio||1,2);W=window.innerWidth;H=window.innerHeight;
    canvas.width=W*DPR;canvas.height=H*DPR;ctx.setTransform(DPR,0,0,DPR,0,0);
  }
  window.addEventListener('resize',resize);resize();

  let gameState='START',rows=new Map(),particles=[],floatingTexts=[],score=0,runCoins=0,minAllowedRow=-3;
  const player={gridX:0,gridY:0,x:0,y:0,startX:0,startY:0,targetX:0,targetY:0,jumping:false,jumpTimer:0,jumpDuration:.13,facing:0,dead:false};
  const camera={x:0,y:0};

  function createRow(index){
    if(index<=1){
      const trees=new Set();
      for(let c=-HALF_COLS;c<=HALF_COLS;c++){if(index<=-3||Math.abs(c)===HALF_COLS)trees.add(c)}
      return{index,type:'grass',trees,coin:null};
    }
    const diff=Math.min(index/65,1),prev=rows.get(index-1),mode=saveData.mapMode;
    let type='grass';
    if(mode==='road'){
      type=(index%5===0)?'grass':'road';
    }else if(mode==='river'){
      type=(index%5===0)?'grass':'river';
    }else if(mode==='rail'){
      type=(index%4===0)?'grass':'rail';
    }else{
      const r=Math.random();
      if(prev&&prev.type==='rail')type='grass';
      else if(r<.25)type='grass';
      else if(r<.65)type='road';
      else if(r<.89)type='river';
      else type='rail';
    }

    let coin=null;
    if(Math.random()<.3){
      const cc=Math.floor(Math.random()*(COLS-2))-(HALF_COLS-1);
      coin={col:cc,x:cc*TILE,collected:false,phase:Math.random()*Math.PI*2};
    }

    if(type==='grass'){
      const trees=new Set(),cnt=2+Math.floor(Math.random()*3);
      for(let i=0;i<cnt;i++){
        const c=Math.floor(Math.random()*COLS)-HALF_COLS;
        if(coin&&coin.col===c)continue;
        if(Math.abs(c)<=1&&Math.random()<.65)continue;
        trees.add(c);
      }
      return{index,type:'grass',trees,coin};
    }
    if(type==='road'){
      const dir=Math.random()<.5?1:-1;
      const speed=dir*(85+Math.random()*110)*(1+diff*.75);
      const count=2+(Math.random()<.45+diff*.3?1:0);
      const span=(COLS+6)*TILE,spacing=span/count,cars=[];
      const palette=['#ef4444','#3b82f6','#f59e0b','#8b5cf6','#ec4899','#14b8a6'];
      for(let i=0;i<count;i++){
        const isTruck=Math.random()<.25;
        cars.push({x:-span/2+i*spacing+Math.random()*40,w:isTruck?TILE*1.95:TILE*1.25,color:palette[Math.floor(Math.random()*palette.length)],isTruck});
      }
      return{index,type:'road',speed,cars,span,coin};
    }
    if(type==='river'){
      let dir=Math.random()<.5?1:-1;
      if(prev&&prev.type==='river')dir=-prev.dir;
      const speed=dir*(65+Math.random()*75)*(1+diff*.5);
      const count=3,span=(COLS+7)*TILE,spacing=span/count,logs=[];
      for(let i=0;i<count;i++)logs.push({x:-span/2+i*spacing+Math.random()*30,w:TILE*(2.1+Math.random()*1.1)});
      return{index,type:'river',dir,speed,logs,span,coin};
    }
    if(type==='rail'){
      const dir=Math.random()<.5?1:-1;
      return{index,type:'rail',dir,timer:1.2+Math.random()*2.5,interval:4.0-diff*1.1,warning:false,trainX:null,coin};
    }
  }

  function ensureRowsAround(center){
    for(let r=center-12;r<=center+18;r++)if(!rows.has(r))rows.set(r,createRow(r));
    for(const k of rows.keys())if(k<center-20)rows.delete(k);
  }

  function resetGame(){
    rows.clear();particles=[];floatingTexts=[];score=0;runCoins=0;minAllowedRow=-3;
    player.gridX=0;player.gridY=0;player.x=0;player.y=0;player.startX=0;player.startY=0;
    player.targetX=0;player.targetY=0;player.jumping=false;player.jumpTimer=0;player.facing=0;player.dead=false;
    ensureRowsAround(0);camera.x=0;camera.y=0;updateHUD();
  }

  function updateHUD(){
    document.getElementById('scoreDisplay').textContent=score;
    document.getElementById('bestDisplay').textContent=saveData.bestScore;
    document.getElementById('coinDisplay').textContent=saveData.coins;
    document.getElementById('startCoins').textContent=saveData.coins;
    document.getElementById('overCoins').textContent=saveData.coins;
    const m=MAPS.find(x=>x.id===saveData.mapMode)||MAPS[0];
    document.getElementById('modeBadge').textContent=`${m.icon} ${m.name}`;
  }

  function spawnParticles(x,y,color,count=14){
    for(let i=0;i<count;i++){
      const a=Math.random()*Math.PI*2,s=40+Math.random()*140;
      particles.push({x,y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,vz:80+Math.random()*120,z:8,color,life:.45+Math.random()*.25,maxLife:.7,size:4+Math.random()*4});
    }
  }
  function spawnFloatingText(x,y,text,color='#fde047'){floatingTexts.push({x,y,text,color,life:.85})}

  function tryMove(dx,dy){
    if(gameState!=='PLAYING'||player.dead||player.jumping)return;
    if(dx>0)player.facing=1;else if(dx<0)player.facing=3;else if(dy>0)player.facing=0;else if(dy<0)player.facing=2;
    const ny=player.gridY+dy;
    if(ny<minAllowedRow)return;
    const cx=Math.round(player.x/TILE),nx=cx+dx;
    if(nx<-HALF_COLS||nx>HALF_COLS)return;
    ensureRowsAround(ny);
    const tr=rows.get(ny);
    if(tr&&tr.type==='grass'&&tr.trees.has(nx))return;
    player.gridX=nx;player.gridY=ny;player.startX=player.x;player.startY=player.y;
    player.targetX=nx*TILE;player.targetY=ny*TILE;player.jumping=true;player.jumpTimer=0;
    sfx.jump();
    if(ny>score){
      score=ny;minAllowedRow=Math.max(minAllowedRow,score-5);
      if(score>saveData.bestScore){saveData.bestScore=score;persistSave()}
      updateHUD();
    }
  }

  function triggerGameOver(title,sub,color){
    if(player.dead)return;
    player.dead=true;gameState='GAMEOVER';
    spawnParticles(player.x,player.y,color,22);persistSave();updateHUD();
    setTimeout(()=>{
      document.getElementById('deathReason').textContent=title;
      document.getElementById('deathSub').textContent=sub;
      document.getElementById('finalScore').textContent=score;
      document.getElementById('runCoins').textContent=runCoins;
      document.getElementById('finalBest').textContent=saveData.bestScore;
      renderMenus();
      document.getElementById('gameOverOverlay').classList.remove('hidden');
    },420);
  }

  function update(dt){
    for(let i=particles.length-1;i>=0;i--){
      const p=particles[i];p.life-=dt;p.x+=p.vx*dt;p.y+=p.vy*dt;p.z=Math.max(0,p.z+p.vz*dt);p.vz-=420*dt;
      if(p.life<=0)particles.splice(i,1);
    }
    for(let i=floatingTexts.length-1;i>=0;i--){
      const f=floatingTexts[i];f.life-=dt;f.y+=36*dt;
      if(f.life<=0)floatingTexts.splice(i,1);
    }
    if(gameState!=='PLAYING')return;

    if(player.jumping){
      player.jumpTimer+=dt;
      const t=Math.min(player.jumpTimer/player.jumpDuration,1),ease=1-Math.pow(1-t,2);
      player.x=player.startX+(player.targetX-player.startX)*ease;
      player.y=player.startY+(player.targetY-player.startY)*ease;
      if(t>=1){player.jumping=false;player.x=player.targetX;player.y=player.targetY}
    }

    for(const row of rows.values()){
      if(row.coin&&!row.coin.collected)row.coin.phase+=dt*4.5;
      if(row.type==='road'){
        for(const c of row.cars){
          c.x+=row.speed*dt;const hs=row.span/2;
          if(row.speed>0&&c.x>hs)c.x=-hs;if(row.speed<0&&c.x<-hs)c.x=hs;
        }
      }else if(row.type==='river'){
        for(const l of row.logs){
          l.x+=row.speed*dt;const hs=row.span/2;
          if(row.speed>0&&l.x>hs)l.x=-hs;if(row.speed<0&&l.x<-hs)l.x=hs;
        }
      }else if(row.type==='rail'){
        row.timer-=dt;
        if(row.trainX!==null){
          row.trainX+=row.dir*1350*dt;
          if(Math.abs(row.trainX)>TILE*18){row.trainX=null;row.warning=false;row.timer=row.interval}
        }else if(row.timer<=.85&&!row.warning){
          row.warning=true;if(Math.abs(row.index-player.gridY)<=7)sfx.train();
        }else if(row.timer<=0){row.trainX=-row.dir*TILE*14}
      }
    }

    const cur=rows.get(player.gridY);
    if(cur&&!player.dead){
      if(cur.coin&&!cur.coin.collected&&Math.abs(player.x-cur.coin.x)<TILE*.65){
        cur.coin.collected=true;saveData.coins+=1;runCoins+=1;persistSave();updateHUD();sfx.coin();
        spawnParticles(cur.coin.x,player.y,'#facc15',10);spawnFloatingText(cur.coin.x,player.y+15,'+1 🪙');
      }
      if(cur.type==='road'){
        for(const c of cur.cars){
          if(Math.abs(player.x-c.x)<c.w*.44+TILE*.28){sfx.crash();triggerGameOver('💥 哎呀！被車撞到了','過馬路前記得左右看清車流喔！','#ef4444');return}
        }
      }
      if(cur.type==='rail'&&cur.trainX!==null){
        if(Math.abs(player.x-cur.trainX)<TILE*4.8){sfx.crash();triggerGameOver('🚄 砰！被特快車撞飛了','紅燈閃爍時千萬別闖越平交道！','#f97316');return}
      }
      if(cur.type==='river'&&!player.jumping){
        let onLog=null;
        for(const l of cur.logs)if(Math.abs(player.x-l.x)<l.w*.48){onLog=l;break}
        if(onLog){
          player.x+=cur.speed*dt;player.targetX=player.x;
          if(Math.abs(player.x)>(HALF_COLS+.85)*TILE){sfx.splash();triggerGameOver('🌊 漂出河道落水啦！','記得在漂出邊界前跳上岸！','#38bdf8');return}
        }else{sfx.splash();triggerGameOver('💦 噗通！掉進河裡了','要踩在漂流木上才能安全渡河喔！','#38bdf8');return}
      }
    }
    camera.x+=(player.x-camera.x)*Math.min(1,10*dt);
    camera.y+=(player.y-camera.y)*Math.min(1,10*dt);
  }

  function worldToScreen(wx,wy){return{x:W/2+(wx-camera.x),y:H*.62-(wy-camera.y)}}
  function drawBox(cx,cy,w,d,h,topC,frontC){
    const x=cx-w/2,y=cy-d/2;
    ctx.fillStyle=frontC;ctx.fillRect(x,y+d-h,w,h);
    ctx.fillStyle=topC;ctx.fillRect(x,y-h,w,d);
  }

  function drawCharacterModel(tctx,id,cx,cy,zOff,facing,sx,sy){

    tctx.translate(0,-zOff);tctx.scale(sx,sy);
    const dx=facing===1?1:(facing===3?-1:0);

    if(id==='chicken'){
      tctx.fillStyle='#e2e8f0';tctx.fillRect(-13,-16,26,22);
      tctx.fillStyle='#fff';tctx.fillRect(-13,-22,26,18);
      tctx.fillStyle='#f1f5f9';tctx.fillRect(-16,-12,4,10);tctx.fillRect(12,-12,4,10);
      tctx.fillStyle='#ef4444';tctx.fillRect(-4,-28,8,7);
      if(facing!==2){

        tctx.fillStyle='#0f172a';
        if(dx===0){tctx.fillRect(-8,-18,3,3);tctx.fillRect(5,-18,3,3)}else tctx.fillRect(dx*6,-18,3,3);
      }
    }else if(id==='penguin'){
      tctx.fillStyle='#0f172a';tctx.fillRect(-13,-18,26,24);
      tctx.fillStyle='#1e293b';tctx.fillRect(-13,-24,26,18);
      if(facing!==2){
        tctx.fillStyle='#fff';tctx.fillRect(-9+dx*2,-16,18,19);
        tctx.fillStyle='#f59e0b';tctx.fillRect(-5+dx*7,-15,10,5);

    }else if(id==='rabbit'){
      tctx.fillStyle='#e2e8f0';tctx.fillRect(-13,-16,26,22);
      tctx.fillStyle='#fff';tctx.fillRect(-13,-22,26,18);
      tctx.fillStyle='#fff';tctx.fillRect(-10,-36,6,15);tctx.fillRect(4,-36,6,15);
      tctx.fillStyle='#f472b6';tctx.fillRect(-8,-34,2,11);tctx.fillRect(6,-34,2,11);
      if(facing!==2){
        tctx.fillStyle='#fb7185';tctx.fillRect(-2+dx*7,-14,4,3);

      }else{tctx.fillStyle='#f8fafc';tctx.fillRect(-4,-6,8,8)}

      tctx.fillStyle='#ea580c';tctx.fillRect(-13,-16,26,22);

  function renderMenus(){
    ['startMapGrid','overMapGrid'].forEach(gid=>{
      const el=document.getElementById(gid);el.innerHTML='';
      MAPS.forEach(m=>{
        const b=document.createElement('button');
        b.className='map-btn'+(saveData.mapMode===m.id?' active':'');
        b.innerHTML=`<span style="font-size:18px">${m.icon}</span><span>${m.name}</span>`;
        b.onclick=()=>{saveData.mapMode=m.id;persistSave();sfx.jump();resetGame();renderMenus()};
        el.appendChild(b);
      });
    });
    ['startCharGrid','overCharGrid'].forEach(gid=>{
      const el=document.getElementById(gid);el.innerHTML='';
      CHARACTERS.forEach(ch=>{
        const owned=saveData.unlocked.includes(ch.id),sel=saveData.selected===ch.id,can=saveData.coins>=ch.price;
        const card=document.createElement('div');
        card.className='char-card'+(sel?' selected':'')+(!owned?' locked':'');
        const cv=document.createElement('canvas');cv.width=92;cv.height=92;
        const pctx=cv.getContext('2d');pctx.scale(2,2);drawCharacterModel(pctx,ch.id,23,37,0,0,1,1);
        const nm=document.createElement('div');nm.className='char-name';nm.textContent=ch.name;
        const tg=document.createElement('div');
        if(sel){tg.className='char-tag tag-active';tg.textContent='使用中'}
        else if(owned){tg.className='char-tag tag-owned';tg.textContent='已擁有'}
        else{tg.className='char-tag '+(can?'tag-buy':'tag-cant');tg.textContent=`🪙 ${ch.price}`}
        card.append(cv,nm,tg);
        card.onclick=()=>{
          if(owned){saveData.selected=ch.id;persistSave();sfx.jump();renderMenus()}
          else if(can){saveData.coins-=ch.price;saveData.unlocked.push(ch.id);saveData.selected=ch.id;persistSave();updateHUD();sfx.buy();renderMenus()}
          else sfx.crash();
        };
        el.appendChild(card);
      });
    });
  }

  function render(){
    ctx.fillStyle='#1e293b';ctx.fillRect(0,0,W,H);
    const minR=Math.floor((camera.y-H*.55)/TILE)-2,maxR=Math.ceil((camera.y+H*.75)/TILE)+2;
    ensureRowsAround(Math.round(camera.y/TILE));
    const vis=[];for(let r=maxR;r>=minR;r--)if(rows.has(r))vis.push(rows.get(r));
    const totalW=(COLS+16)*TILE;

    for(const row of vis){
      const pos=worldToScreen(0,row.index*TILE),y=pos.y;
      if(row.type==='grass'){
        ctx.fillStyle=row.index%2===0?'#86efac':'#4ade80';
        ctx.fillRect(pos.x-totalW/2,y-TILE/2,totalW,TILE+1);
        ctx.fillStyle='rgba(20,83,45,.22)';const pw=COLS*TILE;
        ctx.fillRect(pos.x-totalW/2,y-TILE/2,(totalW-pw)/2,TILE+1);
        ctx.fillRect(pos.x+pw/2,y-TILE/2,(totalW-pw)/2,TILE+1);
      }else if(row.type==='road'){
        ctx.fillStyle='#334155';ctx.fillRect(pos.x-totalW/2,y-TILE/2,totalW,TILE+1);
        ctx.fillStyle='rgba(248,250,252,.35)';
        for(let sx=-HALF_COLS-6;sx<=HALF_COLS+6;sx++){const dp=worldToScreen(sx*TILE,row.index*TILE);ctx.fillRect(dp.x-10,dp.y-TILE/2-1,20,3)}
      }else if(row.type==='river'){
        ctx.fillStyle='#38bdf8';ctx.fillRect(pos.x-totalW/2,y-TILE/2,totalW,TILE+1);
        ctx.fillStyle='rgba(255,255,255,.25)';const ws=(performance.now()*.03*row.dir)%TILE;
        for(let sx=-HALF_COLS-5;sx<=HALF_COLS+5;sx+=2){const wp=worldToScreen(sx*TILE+ws,row.index*TILE);ctx.fillRect(wp.x-12,wp.y-2,24,4)}
        for(const l of row.logs){
          const lp=worldToScreen(l.x,row.index*TILE);drawBox(lp.x,lp.y,l.w,TILE*.72,6,'#b45309','#78350f');
          ctx.fillStyle='#d97706';ctx.fillRect(lp.x-l.w*.35,lp.y-10,l.w*.7,4);
        }
      }else if(row.type==='rail'){
        ctx.fillStyle='#475569';ctx.fillRect(pos.x-totalW/2,y-TILE/2,totalW,TILE+1);
        for(let sx=-HALF_COLS-8;sx<=HALF_COLS+8;sx++){const tp=worldToScreen(sx*TILE*.6,row.index*TILE);ctx.fillStyle='#78350f';ctx.fillRect(tp.x-5,tp.y-15,10,30)}
        ctx.fillStyle='#cbd5e1';ctx.fillRect(pos.x-totalW/2,y-11,totalW,4);ctx.fillRect(pos.x-totalW/2,y+7,totalW,4);
        const sp=worldToScreen(-HALF_COLS*TILE-18,row.index*TILE);
        ctx.fillStyle='#1e293b';ctx.fillRect(sp.x-3,sp.y-24,6,24);
        const blk=row.warning&&Math.floor(performance.now()/120)%2===0;
        ctx.fillStyle=blk?'#ef4444':'#7f1d1d';ctx.beginPath();ctx.arc(sp.x,sp.y-24,6,0,Math.PI*2);ctx.fill();
      }
      if(row.coin&&!row.coin.collected){
        const cp=worldToScreen(row.coin.x,row.index*TILE),bob=Math.sin(row.coin.phase)*4,sw=Math.max(4,Math.abs(Math.cos(row.coin.phase*.7))*12);
        ctx.fillStyle='rgba(0,0,0,.2)';ctx.beginPath();ctx.ellipse(cp.x,cp.y+8,9,4,0,0,Math.PI*2);ctx.fill();
        ctx.fillStyle='#facc15';ctx.beginPath();ctx.ellipse(cp.x,cp.y-10-bob,sw,12,0,0,Math.PI*2);ctx.fill();
        ctx.strokeStyle='#ca8a04';ctx.lineWidth=2;ctx.stroke();
      }
      if(row.type==='grass'){
        for(const col of row.trees){
          const tp=worldToScreen(col*TILE,row.index*TILE);
          drawBox(tp.x,tp.y,16,16,12,'#78350f','#451a03');
          drawBox(tp.x,tp.y-12,36,34,24,'#22c55e','#15803d');
          drawBox(tp.x,tp.y-28,24,22,14,'#4ade80','#16a34a');
        }
      }
      if(row.type==='road'){
        for(const c of row.cars){
          const cp=worldToScreen(c.x,row.index*TILE);drawBox(cp.x,cp.y,c.w,TILE*.68,16,c.color,'#0f172a');
          ctx.fillStyle='#e0f2fe';const cw=c.isTruck?c.w*.32:c.w*.52,co=c.isTruck?(row.speed>0?c.w*.25:-c.w*.25):0;
          ctx.fillRect(cp.x-cw/2+co,cp.y-24,cw,14);
        }
      }
      if(row.type==='rail'&&row.trainX!==null){
        const trp=worldToScreen(row.trainX,row.index*TILE);drawBox(trp.x,trp.y,TILE*9.5,TILE*.78,26,'#ef4444','#991b1b');
        ctx.fillStyle='#fef08a';for(let wx=-TILE*3.8;wx<=TILE*3.8;wx+=TILE*1.1)ctx.fillRect(trp.x+wx-12,trp.y-24,24,10);
      }
      if(row.index===player.gridY&&!player.dead){
        const pp=worldToScreen(player.x,player.y);let z=0,sx=1,sy=1;
        if(player.jumping){const t=player.jumpTimer/player.jumpDuration;z=Math.sin(t*Math.PI)*24;sy=1+Math.sin(t*Math.PI)*.22;sx=1-Math.sin(t*Math.PI)*.14}
        drawCharacterModel(ctx,saveData.selected,pp.x,pp.y,z,player.facing,sx,sy);
      }
    }
    for(const p of particles){const sp=worldToScreen(p.x,p.y);ctx.fillStyle=p.color;ctx.globalAlpha=Math.max(0,p.life/p.maxLife);ctx.fillRect(sp.x-p.size/2,sp.y-p.z-p.size/2,p.size,p.size)}
    ctx.globalAlpha=1;ctx.font='900 18px sans-serif';ctx.textAlign='center';
    for(const f of floatingTexts){const sp=worldToScreen(f.x,f.y);ctx.globalAlpha=Math.min(1,f.life*1.5);ctx.fillStyle='#0f172a';ctx.fillText(f.text,sp.x+1,sp.y-19);ctx.fillStyle=f.color;ctx.fillText(f.text,sp.x,sp.y-20)}
    ctx.globalAlpha=1;
  }

    if(gameState!=='PLAYING'){if(e.code==='Space'||e.code==='Enter')startGame();return}
    if(e.key==='ArrowUp'||e.key==='w'||e.key==='W')tryMove(0,1);
    else if(e.key==='ArrowDown'||e.key==='s'||e.key==='S')tryMove(0,-1);
    else if(e.key==='ArrowLeft'||e.key==='a'||e.key==='A')tryMove(-1,0);
    else if(e.key==='ArrowRight'||e.key==='d'||e.key==='D')tryMove(1,0);
  });

  let tsx=0,tsy=0;
  canvas.addEventListener('pointerdown',e=>{if(gameState==='PLAYING'){tsx=e.clientX;tsy=e.clientY}});
  canvas.addEventListener('pointerup',e=>{
    if(gameState!=='PLAYING')return;
    const dx=e.clientX-tsx,dy=e.clientY-tsy;
    if(Math.hypot(dx,dy)<18)tryMove(0,1);
    else if(Math.abs(dx)>Math.abs(dy))tryMove(dx>0?1:-1,0);
    else tryMove(0,dy<0?1:-1);
  });

  document.querySelectorAll('.dpad-btn').forEach(b=>b.addEventListener('pointerdown',e=>{
    e.stopPropagation();const d=b.dataset.dir;
    if(d==='up')tryMove(0,1);if(d==='down')tryMove(0,-1);if(d==='left')tryMove(-1,0);if(d==='right')tryMove(1,0);
  }));

  resetGame();renderMenus();requestAnimationFrame(loop);
})();
